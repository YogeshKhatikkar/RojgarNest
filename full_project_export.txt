#!/usr/bin/env bash
# ============================================================
# SEVANEXT — BACKUP SCRIPT
# ============================================================
# Usage:
#   ./backup.sh
#
# Cron (daily at 2 AM):
#   0 2 * * * cd /opt/sevanext/deploy && ./backup.sh >> /var/log/sevanext-backup.log 2>&1
# ============================================================
set -euo pipefail

DEPLOY_DIR="$(cd "$(dirname "$0")" && pwd)"
BACKUP_DIR="/opt/sevanext/backups"
TS=$(date +%Y%m%d_%H%M%S)
mkdir -p "$BACKUP_DIR"

cd "$DEPLOY_DIR"

# Resolve actual container IDs (works regardless of COMPOSE_PROJECT_NAME)
MONGO_C=$(docker compose \
  -f docker-compose.yml -f docker-compose.prod.yml \
  --env-file .env.prod ps -q mongo 2>/dev/null || true)

REDIS_C=$(docker compose \
  -f docker-compose.yml -f docker-compose.prod.yml \
  --env-file .env.prod ps -q redis 2>/dev/null || true)

# ─── Mongo backup ───
if [ -n "$MONGO_C" ]; then
  echo "📦 Backing up MongoDB ($MONGO_C)..."
  docker exec "$MONGO_C" mongodump --db=SevaNext --archive --gzip \
    > "$BACKUP_DIR/mongo_${TS}.archive.gz" 2>/dev/null || true
else
  echo "⚠️  MongoDB container not running — skipping"
fi

# ─── Redis backup ───
if [ -n "$REDIS_C" ]; then
  echo "📦 Backing up Redis ($REDIS_C)..."
  docker exec "$REDIS_C" redis-cli SAVE >/dev/null 2>&1 || true
  docker cp "$REDIS_C":/data/dump.rdb \
    "$BACKUP_DIR/redis_${TS}.rdb" 2>/dev/null || true
else
  echo "⚠️  Redis container not running — skipping"
fi

# ─── Retention: delete files older than 7 days ───
find "$BACKUP_DIR" -type f -mtime +7 -delete

echo "✅ Backup complete: $BACKUP_DIR"
ls -lh "$BACKUP_DIR" | tail -5

#!/usr/bin/env bash

#   ./run-local.sh --no-nuke    # keep volumes, rebuild containers only

set -euo pipefail

NO_NUKE=false
[[ "${1:-}" == "--no-nuke" ]] && NO_NUKE=true

C_CYAN='\033[0;36m'; C_YEL='\033[1;33m'; C_GRN='\033[0;32m'; C_RED='\033[0;31m'; C_NC='\033[0m'
hdr() { echo -e "\n${C_CYAN}================================================================${C_NC}"; echo -e "  $1"; echo -e "${C_CYAN}================================================================${C_NC}"; }
step() { echo -e "${C_YEL}[$(date +%H:%M:%S)] $1${C_NC}"; }
ok()   { echo -e "  ${C_GRN}✔ $1${C_NC}"; }
warn() { echo -e "  ${C_YEL}⚠ $1${C_NC}"; }
err()  { echo -e "  ${C_RED}✘ $1${C_NC}"; }

hdr "SEVANEXT - LOCAL DOCKER FULL RESET & DEPLOY"

[[ -f docker-compose.yml ]] || { err "docker-compose.yml missing"; exit 1; }
[[ -f .env.local ]]         || { err ".env.local missing"; exit 1; }
[[ -f ../backend/sevanext/.env ]] || warn "../backend/sevanext/.env missing"

COMPOSE=(docker compose -f docker-compose.yml -f docker-compose.dev.yml --env-file .env.local)

# ---- STEP 1: NUKE ----
if [ "$NO_NUKE" = false ]; then
    step "[1/8] FULL NUKE — removing containers, images, volumes, networks, build cache..."
    docker compose down -v --remove-orphans 2>/dev/null || true
    docker compose -f docker-compose.yml -f docker-compose.prod.yml --env-file .env.prod down -v --remove-orphans 2>/dev/null || true

    echo "  -> removing ALL containers"
    docker ps -aq | xargs -r docker rm -f 2>/dev/null || true

    echo "  -> removing ALL images"
    docker images -q | xargs -r docker rmi -f 2>/dev/null || true

    echo "  -> removing ALL volumes"
    docker volume ls -q | xargs -r docker volume rm -f 2>/dev/null || true

    echo "  -> pruning networks"
    docker network prune -f 2>/dev/null || true

    echo "  -> pruning build cache"
    docker builder prune -af 2>/dev/null || true

    echo "  -> pruning system"
    docker system prune -af --volumes 2>/dev/null || true

    ok "FULL NUKE complete."
else
    warn "[1/8] --no-nuke: keeping volumes & images"
    "${COMPOSE[@]}" down --remove-orphans 2>/dev/null || true
fi

# ---- STEP 2: DOCKER DAEMON ----
step "[2/8] Verifying Docker daemon..."
docker info >/dev/null 2>&1 || { err "Docker daemon not running."; exit 1; }
ok "Docker daemon running."

step "[3/8] Building backend + frontend images..."
BUILD_ARGS=()
[ "$NO_NUKE" = false ] && BUILD_ARGS+=(--no-cache)
"${COMPOSE[@]}" build "${BUILD_ARGS[@]}" || { err "build FAILED — scroll up"; exit 1; }
ok "Images built."

# ---- STEP 4: START ----
step "[4/8] Starting stack..."
"${COMPOSE[@]}" up -d --remove-orphans || { err "up FAILED"; "${COMPOSE[@]}" logs --tail=80; exit 1; }
ok "Containers launched."

E=0; OK=false
while [ $E -lt 120 ]; do
    S=$(docker inspect --format='{{.State.Health.Status}}' sevanext-local-mongo 2>/dev/null || echo "starting")
    [ "$S" = "healthy" ] && { OK=true; break; }
    sleep 3; E=$((E+3)); echo "      [$E s] mongo: $S"
done
$OK || { err "MongoDB not healthy after 120s."; docker logs --tail 60 sevanext-local-mongo; exit 1; }
ok "MongoDB healthy."

# ---- STEP 6: WAIT FOR BACKEND ----
step "[6/8] Waiting for backend /api/v1/health (max 300s)..."
E=0; OK=false
while [ $E -lt 300 ]; do
    S=$(docker inspect --format='{{.State.Health.Status}}' sevanext-local-backend 2>/dev/null || echo "starting")
    if [ "$S" = "healthy" ]; then OK=true; break; fi
    if [ "$S" = "unhealthy" ]; then err "Backend UNHEALTHY"; docker logs --tail 100 sevanext-local-backend; exit 1; fi
    sleep 3; E=$((E+3)); echo "      [$E s] backend: $S"
done
if ! $OK; then
    warn "Backend not healthy after 300s — showing last 100 logs:"
    docker logs --tail 100 sevanext-local-backend || true
    warn "Common fixes: check ../backend/sevanext/.env has MONGO_URI, SECRET_KEY, JWT_SECRET_KEY"

def sanitize_env_file(content: str) -> str:
    """
    Sanitize .env file content by replacing actual values with dummy values.
    Original values ko hide karta hai, dummy values show karta hai.
    """
    lines = content.split('\n')
    sanitized_lines = []
    
    for line in lines:
        # Remove leading/trailing whitespace
        line = line.rstrip('\n\r')
        
        # Skip empty lines
        if not line.strip():
            sanitized_lines.append(line)
            continue
        
        # Skip comments
        if line.strip().startswith('#'):
            sanitized_lines.append(line)
            continue
        
        # Check if line has KEY=VALUE format
        if '=' in line:
            # Split at first '='
            parts = line.split('=', 1)
            key = parts[0].strip()
            value = parts[1].strip() if len(parts) > 1 else ''
            
            # Apply sanitization if key is in mapping
            if key in ENV_SANITIZATION_RULES:
                dummy_value = ENV_SANITIZATION_RULES[key]
                # Preserve quotes if original had quotes
                if value.startswith('"') and value.endswith('"'):
                    sanitized_lines.append(f'{key}="{dummy_value}"')
                elif value.startswith("'") and value.endswith("'"):
                    sanitized_lines.append(f"{key}='{dummy_value}'")
                else:
                    sanitized_lines.append(f'{key}={dummy_value}')
            else:
                # Unknown key - put placeholder
                sanitized_lines.append(f'{key}=[REDACTED_ADD_TO_ENV_SANITIZATION_RULES]')
        else:
            # Malformed line - keep as is
            sanitized_lines.append(line)
    
    # Add warning header at top of .env content
    warning_header = """# ============================================================
# ⚠️  WARNING: THIS IS A SANITIZED .env FILE
# All original values have been replaced with dummy/placeholder values.
# Add your actual credentials in the original .env file.
# ============================================================

"""
    
    return warning_header + '\n'.join(sanitized_lines)

def get_file_encoding(file_path: Path) -> str:
    """Detect file encoding"""
    try:
        with open(file_path, 'rb') as f:
            raw = f.read(1000)
            if raw.startswith(b'\xef\xbb\xbf'):
                return 'utf-8-sig'
            if b'\0' in raw:
                return 'utf-16'
    except:
        pass
    return 'utf-8'

                try:

def should_include_file(file_path: Path) -> bool:
    """Check if file should be included in export"""
    file_name = file_path.name
    
    # Force include specific files
    if file_name in FORCE_INCLUDE:
        return True
    
    # Check file size
    try:
        if file_path.stat().st_size > MAX_FILE_SIZE:
            return False
    except:
        pass
    
    # Check extension
    if file_path.suffix not in EXTENSIONS:
        return False
    
    # Check exclude list (with pattern matching)
    for pattern in EXCLUDE_FILES:
        if '*' in pattern:
            import fnmatch
            if fnmatch.fnmatch(file_name, pattern):
                return False
        elif file_name == pattern:
            return False
    
    # Skip empty files
    try:
        if file_path.stat().st_size == 0:
            return False
    except:
        pass
    
    return True

def read_file_content(file_path: Path) -> Tuple[str, str]:
    """Read file content with proper encoding"""
    # Try different encodings
    encodings = ['utf-8', 'utf-8-sig', 'latin-1', 'cp1252', 'iso-8859-1']
    
    for encoding in encodings:
        try:
            with open(file_path, 'r', encoding=encoding) as f:
                content = f.read()
            return content, encoding
        except UnicodeDecodeError:
            continue
        except Exception as e:
            continue
    
    # If all encodings fail, try with replacement
    try:
        with open(file_path, 'r', encoding='utf-8', errors='replace') as f:
            content = f.read()
        return content, 'utf-8 (with replacements)'
    except Exception as e:
        return f"[ERROR: Could not read file - {str(e)}]", 'error'

def normalize(path: Path) -> str:

def is_excluded_folder(folder_path: Path) -> Tuple[bool, str]:
    name = folder_path.name.lower()
    if name in EXCLUDE_FOLDERS:

def is_excluded_file(file_path: Path) -> Tuple[bool, str]:
    name = file_path.name
    name_lower = name.lower()

    try:
        size = file_path.stat().st_size
        if size > MAX_FILE_SIZE:
            return True, f"too large ({size / (1024*1024):.1f} MB)"
    except Exception as e:
        return True, f"stat error: {e}"

def scan_folder(folder: Path, root: Path) -> Tuple[List[Dict], List[Dict]]:
    included, skipped = [], []

    def walk(current: Path):
        try:
            for item in sorted(current.iterdir()):
                try:
                    rel = item.relative_to(root)
                except ValueError:
                    rel = item

                if not item.exists():
                    continue

                is_dir = item.is_dir() and not item.is_symlink()
                is_file = item.is_file() or (item.is_symlink() and item.resolve().is_file())

                if is_dir:
                    ex, reason = is_excluded_folder(item)
                    if ex:
                        skipped.append({'path': item, 'rel_path': rel,
                                        'reason': reason, 'is_dir': True})
                    else:
                        walk(item)

                elif is_file:
                    ex, reason = is_excluded_file(item)
                    entry = {'path': item, 'rel_path': rel,
                             'reason': reason, 'is_dir': False}
                    if ex:
                        skipped.append(entry)
                    else:
                        included.append(entry)

        except PermissionError as e:
            print(f"⚠️ Permission denied: {current} — {e}")
        except Exception as e:
            print(f"⚠️ Error scanning {current}: {e}")

    walk(folder)
    return included, skipped


def generate_tree(folder: Path, root: Path, prefix: str = "") -> List[str]:
    lines = []
    try:
        items = []
        for item in sorted(folder.iterdir()):
            if item.is_dir() and not item.is_symlink():
                ex, _ = is_excluded_folder(item)
                if not ex:
                    items.append(item)
            elif item.is_file():
                ex, _ = is_excluded_file(item)
                if not ex:
                    items.append(item)
        for i, item in enumerate(items):
            last = (i == len(items) - 1)
            conn = "└── " if last else "├── "
            if item.is_dir():
                lines.append(f"{prefix}{conn}📁 {item.name}/")
                ext = "    " if last else "│   "
                lines.extend(generate_tree(item, root, prefix + ext))
            else:
                lines.append(f"{prefix}{conn}{get_file_icon(item)} {item.name}")
    except Exception as e:
        lines.append(f"{prefix}[Error: {e}]")
    return lines

REPLACEMENTS = [

def is_binary(file_path: Path) -> bool:
    """Check if file is binary — skip those."""
    try:
        with open(file_path, "rb") as f:
            chunk = f.read(8192)
            return b"\0" in chunk
    except Exception:
        return True


def should_skip_dir(dir_name: str) -> bool:
    return dir_name in SKIP_DIRS


def should_skip_file(file_name: str) -> bool:
    return file_name in SKIP_FILES


def find_files(root: Path, extensions: set) -> list:
    """Recursively find all matching files, skipping skip-dirs."""
    if not root.exists():
        print(f"   ⚠️  Directory not found: {root}")
        return []

    found = []
    for dirpath, dirnames, filenames in os.walk(root):
        # Prune skip dirs in-place
        dirnames[:] = [d for d in dirnames if not should_skip_dir(d)]

        for filename in filenames:
            if should_skip_file(filename):
                continue

            file_path = Path(dirpath) / filename

            # Extension check (allow files with no extension like `.env` or `Dockerfile`)
            if file_path.suffix.lower() in extensions:
                found.append(file_path)
            elif filename in ("Dockerfile", ".env", ".env.example", ".dockerignore", "nginx.conf"):
                found.append(file_path)

    return found


def count_matches(content: str) -> dict:
    """Count how many times each pattern appears."""
    counts = {}
    for old, _ in REPLACEMENTS:
        n = content.count(old)
        if n > 0:
            counts[old] = n
    return counts


def apply_replacements(content: str) -> tuple:
    """Apply all replacements in order. Returns (new_content, total_changes)."""
    total = 0
    new = content
    for old, replacement in REPLACEMENTS:
        count = new.count(old)
        if count > 0:
            new = new.replace(old, replacement)
            total += count
    return new, total


def backup_file(file_path: Path) -> Path:
    """Copy file to backup folder preserving relative structure."""
    try:
        rel = file_path.relative_to(PROJECT_ROOT)
    except ValueError:
        rel = Path(file_path.name)

    backup_path = BACKUP_ROOT / rel
    backup_path.parent.mkdir(parents=True, exist_ok=True)

    # Don't overwrite existing backups
    if not backup_path.exists():
        shutil.copy2(file_path, backup_path)

    return backup_path

def process_root(root: Path, extensions: set, label: str, dry_run: bool):
    """Process one root folder (backend or frontend)."""
    print(f"\n{'='*70}")
    print(f"📂 {label}: {root}")
    print(f"{'='*70}")

    if not root.exists():
        print(f"   ❌ Folder does not exist. Skipping.")
        return {"scanned": 0, "changed": 0, "total_replacements": 0, "files": []}

    files = find_files(root, extensions)
    print(f"   🔍 Found {len(files)} candidate files")

    scanned = 0
    changed_files = []
    total_replacements = 0

    for f in files:
        scanned += 1

        # Skip binary
        if is_binary(f):
            continue

        try:
            original = f.read_text(encoding="utf-8")
        except UnicodeDecodeError:
            try:
                original = f.read_text(encoding="latin-1")
            except Exception:
                continue
        except Exception as e:
            print(f"   ⚠️  Could not read {f.relative_to(PROJECT_ROOT)}: {e}")
            continue

        # Count matches BEFORE replacing
        matches = count_matches(original)
        if not matches:
            continue

        new_content, n = apply_replacements(original)
        if n == 0:
            continue

        rel_path = f.relative_to(PROJECT_ROOT)
        print(f"\n   📝 {rel_path}")
        for pattern, cnt in matches.items():
            print(f"       • '{pattern}' → {cnt} occurrence(s)")

        changed_files.append({
            "file": str(rel_path),
            "replacements": n,
            "details": matches,
        })
        total_replacements += n

        if not dry_run:
            backup_file(f)
            try:
                f.write_text(new_content, encoding="utf-8")
                print(f"       ✅ Updated (backup saved)")
            except Exception as e:
                print(f"       ❌ Write failed: {e}")

    return {
        "scanned": scanned,
        "changed": len(changed_files),
        "total_replacements": total_replacements,
        "files": changed_files,
    }

def do_rollback():
    """Find latest backup folder and restore."""
    print(f"\n{'='*70}")
    print(f"↩️  ROLLBACK")
    print(f"{'='*70}")

    # Find all backup folders
    backups = sorted(PROJECT_ROOT.glob("_backup_*"))

    if not backups:
        print("   ❌ No backup folders found.")
        return

    latest = backups[-1]
    print(f"   📁 Using backup: {latest.name}")

    restored = 0
    for root_dir in (BACKEND_DIR, FRONTEND_DIR):
        backup_dir = latest / "backend" / "sevanext" if "backend" in str(root_dir).lower() else latest / "frontend" / "sevanext"

        # Actually reconstruct properly:
        if root_dir == BACKEND_DIR:
            backup_dir = latest / "backend" / "sevanext"
        elif root_dir == FRONTEND_DIR:
            backup_dir = latest / "frontend" / "sevanext"

        if not backup_dir.exists():
            continue

        for backup_file_path in backup_dir.rglob("*"):
            if backup_file_path.is_dir():
                continue

            rel = backup_file_path.relative_to(backup_dir)
            original = root_dir / rel

            original.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(backup_file_path, original)
            print(f"   ♻️  Restored: {rel}")
            restored += 1

    print(f"\n   ✅ Restored {restored} files")
