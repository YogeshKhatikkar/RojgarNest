#!/usr/bin/env bash
# ============================================================
# ROJGARNEST — DEPLOYMENT HELPER
# ============================================================
set -euo pipefail

cd "$(dirname "$0")"

# Colors
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
CYAN='\033[0;36m'
NC='\033[0m'

# ============================================================
# HELPERS
# ============================================================
log()      { echo -e "${CYAN}[deploy]${NC} $*"; }
success()  { echo -e "${GREEN}[ok]${NC} $*"; }
warn()     { echo -e "${YELLOW}[warn]${NC} $*"; }
error()    { echo -e "${RED}[error]${NC} $*" >&2; exit 1; }

# ============================================================
# COMMANDS
# ============================================================
cmd_dev_up() {
    log "Starting DEVELOPMENT stack..."
    docker compose \
        -f docker-compose.yml \
        -f docker-compose.dev.yml \
        --env-file .env.dev \
        up --build -d
    success "Development stack up. Frontend: http://localhost:8080 | Backend: http://localhost:8000"
}

cmd_dev_down() {
    log "Stopping DEVELOPMENT stack..."
    docker compose \
        -f docker-compose.yml \
        -f docker-compose.dev.yml \
        --env-file .env.dev \
        down
    success "Development stack down"
}

cmd_dev_logs() {
    docker compose \
        -f docker-compose.yml \
        -f docker-compose.dev.yml \
        --env-file .env.dev \
        logs -f
}

cmd_prod_up() {
    log "Starting PRODUCTION stack..."
    [ -f .env.prod ] || error ".env.prod not found — copy .env.prod.example"
    docker compose \
        -f docker-compose.yml \
        -f docker-compose.prod.yml \
        --env-file .env.prod \
        up --build -d
    success "Production stack up. Frontend: http://localhost | Backend: internal"
}

cmd_prod_down() {
    log "Stopping PRODUCTION stack..."
    docker compose \
        -f docker-compose.yml \
        -f docker-compose.prod.yml \
        --env-file .env.prod \
        down
    success "Production stack down"
}

cmd_prod_logs() {
    docker compose \
        -f docker-compose.yml \
        -f docker-compose.prod.yml \
        --env-file .env.prod \
        logs -f
}

cmd_backup_db() {
    log "Backing up MongoDB..."
    local ts=$(date +%Y%m%d_%H%M%S)
    local out="backups/mongo_${ts}.gz"
    mkdir -p backups
    docker compose exec -T mongo mongodump --archive --gzip > "$out"
    success "Backup saved: $out"
}

cmd_restore_db() {
    local file="${1:-}"
    [ -n "$file" ] || error "Usage: ./deploy.sh restore-db <backup.gz>"
    [ -f "$file" ] || error "File not found: $file"
    log "Restoring MongoDB from $file..."
    docker compose exec -T mongo mongorestore --archive --gzip < "$file"
    success "Restore complete"
}

cmd_seed() {
    log "Seeding schemes..."
    docker compose \
        -f docker-compose.yml \
        -f docker-compose.dev.yml \
        --env-file .env.dev \
        exec backend python seed_schemes.py
    success "Seeding complete"
}

cmd_status() {
    docker compose ps
}

cmd_health() {
    log "Checking health..."
    curl -fsS http://localhost:8000/api/v1/health && echo
    curl -fsS http://localhost:8080/healthz && echo
    success "All healthy"
}

# ============================================================
# ROUTER
# ============================================================
case "${1:-help}" in
    dev-up)      cmd_dev_up ;;
    dev-down)    cmd_dev_down ;;
    dev-logs)    cmd_dev_logs ;;
    prod-up)     cmd_prod_up ;;
    prod-down)   cmd_prod_down ;;
    prod-logs)   cmd_prod_logs ;;
    backup-db)   cmd_backup_db ;;
    restore-db)  shift; cmd_restore_db "${1:-}" ;;
    seed)        cmd_seed ;;
    status)      cmd_status ;;
    health)      cmd_health ;;
    *)
        cat <<EOF
Usage: ./deploy.sh <command>

DEV:
  dev-up       Start development stack
  dev-down     Stop development stack
  dev-logs     Follow development logs

PROD:
  prod-up      Start production stack
  prod-down    Stop production stack
  prod-logs    Follow production logs

DB:
  seed         Seed schemes into MongoDB
  backup-db    Backup MongoDB to deploy/backups/
  restore-db   Restore MongoDB from backup file

UTIL:
  status       Show container status
  health       Check endpoints
EOF
        ;;
esac

# ============================================================
# DEV NGINX — Only used if you add an nginx service to dev compose
# For local dev, frontend is served on :8080 directly by its own container.
# You can OPTIONALLY add this as a proxy if you want http://localhost:80 → backend+frontend.

user nginx;
worker_processes auto;
worker_rlimit_nofile 65535;

events {
    worker_connections 2048;
    multi_accept on;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    client_max_body_size 50M;

    gzip on;
    gzip_types text/plain text/css application/json application/javascript
               application/wasm image/svg+xml application/xml;
    gzip_min_length 1024;
    gzip_comp_level 6;

    upstream backend_upstream {
        server backend:8000;
    }

    upstream frontend_upstream {
        server frontend:80;
    }

    server {
        listen 80;
        server_name localhost;

        # ---- Health ----
        location /health {
            access_log off;
            return 200 "healthy\n";
            add_header Content-Type text/plain;
        }

        # ---- API ----
        location /api/ {
            proxy_pass http://backend_upstream;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_read_timeout 120s;
        }

        # ---- WebSocket ----
        location /ws/ {
            proxy_pass http://backend_upstream;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_set_header Host $host;
            proxy_read_timeout 86400s;
        }

        # ---- Backend uploads ----
        location /uploads/ {
            proxy_pass http://backend_upstream;
            expires 1d;
        }

        # ---- Frontend (Flutter web) ----
        location / {
            proxy_pass http://frontend_upstream;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }
    }
}

def sanitize_env_file(content: str) -> str:
    """
    Sanitize .env file content by replacing actual values with dummy values.
    Original values ko hide karta hai, dummy values show karta hai.
    """
    lines = content.split('\n')
    sanitized_lines = []
    
    for line in lines:
        # Remove leading/trailing whitespace
        line = line.rstrip('\n\r')
        
        # Skip empty lines
        if not line.strip():
            sanitized_lines.append(line)
            continue
        
        # Skip comments
        if line.strip().startswith('#'):
            sanitized_lines.append(line)
            continue
        
        # Check if line has KEY=VALUE format
        if '=' in line:
            # Split at first '='
            parts = line.split('=', 1)
            key = parts[0].strip()
            value = parts[1].strip() if len(parts) > 1 else ''
            
            # Apply sanitization if key is in mapping
            if key in ENV_SANITIZATION_RULES:
                dummy_value = ENV_SANITIZATION_RULES[key]
                # Preserve quotes if original had quotes
                if value.startswith('"') and value.endswith('"'):
                    sanitized_lines.append(f'{key}="{dummy_value}"')
                elif value.startswith("'") and value.endswith("'"):
                    sanitized_lines.append(f"{key}='{dummy_value}'")
                else:
                    sanitized_lines.append(f'{key}={dummy_value}')
            else:
                # Unknown key - put placeholder
                sanitized_lines.append(f'{key}=[REDACTED_ADD_TO_ENV_SANITIZATION_RULES]')
        else:
            # Malformed line - keep as is
            sanitized_lines.append(line)
    
    # Add warning header at top of .env content
    warning_header = """# ============================================================
# ⚠️  WARNING: THIS IS A SANITIZED .env FILE
# All original values have been replaced with dummy/placeholder values.
# Add your actual credentials in the original .env file.
# ============================================================

"""
    
    return warning_header + '\n'.join(sanitized_lines)

def get_file_encoding(file_path: Path) -> str:
    """Detect file encoding"""
    try:
        with open(file_path, 'rb') as f:
            raw = f.read(1000)
            if raw.startswith(b'\xef\xbb\xbf'):
                return 'utf-8-sig'
            if b'\0' in raw:
                return 'utf-16'
    except:
        pass
    return 'utf-8'

                scan_directory(item, base_path, all_files)

                    continue
                
                try:

                    # Check if it's .env file

                    })
                except Exception as e:

    except PermissionError:

def should_include_file(file_path: Path) -> bool:
    """Check if file should be included in export"""
    file_name = file_path.name
    
    # Force include specific files
    if file_name in FORCE_INCLUDE:
        return True
    
    # Check file size
    try:
        if file_path.stat().st_size > MAX_FILE_SIZE:
            return False
    except:
        pass
    
    # Check extension
    if file_path.suffix not in EXTENSIONS:
        return False
    
    # Check exclude list (with pattern matching)
    for pattern in EXCLUDE_FILES:
        if '*' in pattern:
            import fnmatch
            if fnmatch.fnmatch(file_name, pattern):
                return False
        elif file_name == pattern:
            return False
    
    # Skip empty files
    try:
        if file_path.stat().st_size == 0:
            return False
    except:
        pass
    
    return True

def read_file_content(file_path: Path) -> Tuple[str, str]:
    """Read file content with proper encoding"""
    # Try different encodings
    encodings = ['utf-8', 'utf-8-sig', 'latin-1', 'cp1252', 'iso-8859-1']
    
    for encoding in encodings:
        try:
            with open(file_path, 'r', encoding=encoding) as f:
                content = f.read()
            return content, encoding
        except UnicodeDecodeError:
            continue
        except Exception as e:
            continue
    
    # If all encodings fail, try with replacement
    try:
        with open(file_path, 'r', encoding='utf-8', errors='replace') as f:
            content = f.read()
        return content, 'utf-8 (with replacements)'
    except Exception as e:

def is_excluded_file(file_path: Path) -> Tuple[bool, str]:
    name = file_path.name

def scan_folder(folder: Path, root: Path) -> Tuple[List[Dict], List[Dict]]:
    included, skipped = [], []

    def walk(current: Path):
        try:
            for item in sorted(current.iterdir()):
                try:
                    rel = item.relative_to(root)
                except ValueError:
                    rel = item

                if not item.exists():
                    continue

                is_dir = item.is_dir() and not item.is_symlink()
                is_file = item.is_file() or (item.is_symlink() and item.resolve().is_file())

                if is_dir:
                    ex, reason = is_excluded_folder(item)
                    if ex:
                        skipped.append({'path': item, 'rel_path': rel,
                                        'reason': reason, 'is_dir': True})
                    else:
                        walk(item)

                elif is_file:
                    ex, reason = is_excluded_file(item)
                    entry = {'path': item, 'rel_path': rel,
                             'reason': reason, 'is_dir': False}
                    if ex:
                        skipped.append(entry)
                    else:
                        included.append(entry)

        except PermissionError as e:
            print(f"⚠️ Permission denied: {current} — {e}")
        except Exception as e:
            print(f"⚠️ Error scanning {current}: {e}")

    walk(folder)
    return included, skipped


def generate_tree(folder: Path, root: Path, prefix: str = "") -> List[str]:
    lines = []
    try:
        items = []
        for item in sorted(folder.iterdir()):
            if item.is_dir() and not item.is_symlink():
                ex, _ = is_excluded_folder(item)
                if not ex:
                    items.append(item)
            elif item.is_file():
                ex, _ = is_excluded_file(item)
                if not ex:
                    items.append(item)
        for i, item in enumerate(items):
            last = (i == len(items) - 1)
            conn = "└── " if last else "├── "
            if item.is_dir():
                lines.append(f"{prefix}{conn}📁 {item.name}/")
                ext = "    " if last else "│   "
                lines.extend(generate_tree(item, root, prefix + ext))
            else:
                lines.append(f"{prefix}{conn}{get_file_icon(item)} {item.name}")
    except Exception as e:
        lines.append(f"{prefix}[Error: {e}]")
    return lines
